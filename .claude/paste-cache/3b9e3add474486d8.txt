# Enable dynamic substitution in the prompt
setopt PROMPT_SUBST

# NUDIMMUD Context Bar Engine
render_context_bar() {
    # ----------------------------------------------------------------------
    # [DATA SOURCE]: Replace the mocked value below with a call to your backend.
    # Example for real integration:
    # local percent=$(curl -s http://localhost:3004/api/metrics/context || echo 0)
    # ----------------------------------------------------------------------
    
    # For now, we are mocking the percentage. Try changing this variable
    # in your terminal (e.g., export CONTEXT_LOAD=85) to see the color shift.
    local percent=${CONTEXT_LOAD:-15} 
    
    # Visual settings
    local total_blocks=20
    local filled_blocks=$(( percent * total_blocks / 100 ))
    local empty_blocks=$(( total_blocks - filled_blocks ))
    
    # ANSI Color Matrix (Zsh format)
    local color="%F{112}" # Default NUDIMMUD Green
    
    if [ "$percent" -ge 85 ]; then
        color="%F{196}"   # Red-Fusion (Critical Load)
    elif [ "$percent" -ge 60 ]; then
        color="%F{214}"   # Warning Orange/Gold-Solar
    fi
    
    local bar_string=""
    
    # Build the filled portion (■)
    for ((i=0; i<filled_blocks; i++)); do
        bar_string+="■"
    done
    
    # Build the empty portion (□)
    for ((i=0; i<empty_blocks; i++)); do
        bar_string+="□"
    done
    
    # Render the formatted string (e.g., "CTX [■■■■□□□□□□] 40%")
    echo "${color}CTX [${bar_string}] ${percent}%%%f"
}

# The precmd hook runs just before the prompt is printed
precmd() {
    # We store the output of the bar in a variable
    PS1_CONTEXT_BAR="$(render_context_bar)"
}

# Update the NUDIMMUD Prompt to stack the bar ABOVE the input line
# %b resets boldness, %f resets color. \n creates a new line.
export PS1=$'\n${PS1_CONTEXT_BAR}\n%F{244}$%f '
